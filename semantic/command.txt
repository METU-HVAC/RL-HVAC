# For different size latent vectors

cd /storage/Master/ParallelRL/RL-HVAC
source ~/.bashrc
conda activate hvac-rl

BASE=/storage/Master/ParallelRL/RL-HVAC/semantic/models/graphs/epoch20_dim
ENV=A403mediumfanger
SEASON=hot

for DIM in 20 24 28 32; do
  python /storage/Master/ParallelRL/RL-HVAC/semantic/train_gnn.py \
    --env-id "$ENV" \
    --season "$SEASON" \
    --chunks 8 \
    --epochs 20 \
    --lr 1e-3 \
    --latent-dim "$DIM"

  OUT_DIR="${BASE}${DIM}/${ENV}_${SEASON}"
  mkdir -p "$OUT_DIR"
  cp /storage/Master/ParallelRL/RL-HVAC/semantic/models/graph_encoder.pt "$OUT_DIR/graph_encoder.pt"
  cp /storage/Master/ParallelRL/RL-HVAC/semantic/models/target_stats.json "$OUT_DIR/target_stats.json"
  cp /storage/Master/ParallelRL/RL-HVAC/semantic/models/graph_config.json "$OUT_DIR/graph_config.json"
done
